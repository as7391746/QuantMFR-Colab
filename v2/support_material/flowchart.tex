% Flow chart of the automatic initial guess (AUTO_GUESS.md), v2.1.
% Compile: pdflatex flowchart.tex ; then rasterize to flowchart.png.
\documentclass[tikz,border=6pt]{standalone}
\usetikzlibrary{arrows.meta,positioning}
\begin{document}
\begin{tikzpicture}[
  font=\small,
  box/.style={rectangle, rounded corners=2pt, draw=black!70, fill=blue!4,
              align=left, inner sep=6pt, text width=8.8cm},
  dec/.style={rectangle, rounded corners=2pt, draw=black!70, very thick,
              fill=orange!8, align=left, inner sep=6pt, text width=8.8cm},
  lp/.style={rectangle, rounded corners=2pt, draw=black!70, fill=black!4,
             align=center, inner sep=5pt, font=\footnotesize, text width=3.2cm},
  a/.style={-{Latex[length=2.6mm]}, semithick},
  lbl/.style={font=\footnotesize, inner sep=2pt, fill=white}
]

\node[box] (decl) {model declaration: $\kappa,\ \psi^g,\ \psi^x,\ \phi$, parameter values};
\node[box, below=4.5mm of decl] (det)
  {set $\mathsf q=0$, set the shock vector to zero, treat the variables as
   time invariant};
\node[box, below=4.5mm of det] (claim)
  {\textbf{What a single equation claims}\\[2pt]
   states from their own evolution equations\\
   one decision from the resource constraint\\
   $g^0$ and $\widehat C^0-\widehat G^0$ by definition;
   $\widehat V^0-\widehat G^0$ in closed form\\
   multipliers by a linear solve};
\node[box, below=4.5mm of claim] (marg)
  {\textbf{Remaining margins} (unclaimed decisions; states whose
   equations degenerate into $\psi^g=$ drift)\\[2pt]
   solved against their own first-order conditions as one small joint
   block, from a few starting values; each candidate rebuilds all other
   entries through the claims above};
\node[dec, below=4.5mm of marg] (land)
  {Does the constructed point satisfy the complete steady-state system?};
\node[lp, right=6mm of land] (fb)
  {fallback: previous construction, grid of trial values (Section 2)};
\node[box, below=4.5mm of land] (ig)
  {\texttt{initial\_guess} assembled --- the constructed steady state
   itself; handed to the root solve of \texttt{uncertain\_expansion},
   which is unchanged};

\draw[a] (decl) -- (det);
\draw[a] (det) -- (claim);
\draw[a] (claim) -- (marg);
\draw[a] (marg) -- (land);
\draw[a] (land.east) -- node[lbl, above]{no} (fb.west);
\draw[a] (land) -- node[lbl, right=1pt]{yes} (ig);
\end{tikzpicture}
\end{document}
